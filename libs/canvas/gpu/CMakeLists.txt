qt_add_resources(gpu_canvas_resources gpu_canvas.qrc)
add_library(kritacanvasgpurenderer STATIC GpuRenderer.cpp ${gpu_canvas_resources})
target_include_directories(kritacanvasgpurenderer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(kritacanvasgpurenderer PUBLIC kritaimagegpudocument Qt${QT_MAJOR_VERSION}::Gui)
set_target_properties(kritacanvasgpurenderer PROPERTIES AUTOMOC OFF POSITION_INDEPENDENT_CODE ON)

if((APPLE AND NOT IOS) OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(kritacanvasgpusurface STATIC SurfaceRenderer.cpp)
    target_include_directories(kritacanvasgpusurface PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(kritacanvasgpusurface PUBLIC kritacanvasgpurenderer)
    set_target_properties(kritacanvasgpusurface PROPERTIES AUTOMOC OFF POSITION_INDEPENDENT_CODE ON)
    add_library(kritacanvasgpudocument STATIC DocumentCanvas.cpp)
    target_include_directories(kritacanvasgpudocument PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(kritacanvasgpudocument PUBLIC kritacanvasgpusurface)
    set_target_properties(kritacanvasgpudocument PROPERTIES AUTOMOC OFF POSITION_INDEPENDENT_CODE ON)
    if(APPLE)
        target_sources(kritacanvasgpusurface PRIVATE GpuWindowSurface_mac.mm)
        find_library(LIBREPAINT_GPU_APPKIT AppKit REQUIRED)
        find_library(LIBREPAINT_GPU_QUARTZCORE QuartzCore REQUIRED)
        target_link_libraries(kritacanvasgpusurface PRIVATE ${LIBREPAINT_GPU_APPKIT} ${LIBREPAINT_GPU_QUARTZCORE})
    else()
        target_sources(kritacanvasgpusurface PRIVATE GpuWindowSurface_linux.cpp)
        find_package(X11 REQUIRED)
        target_link_libraries(kritacanvasgpusurface PRIVATE X11::X11)
        if(TARGET Qt6::WaylandClient)
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(LIBREPAINT_WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
            pkg_check_modules(LIBREPAINT_WAYLAND_PROTOCOLS REQUIRED wayland-protocols)
            pkg_get_variable(LIBREPAINT_WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
            find_program(LIBREPAINT_WAYLAND_SCANNER wayland-scanner REQUIRED)
            foreach(protocol_name IN ITEMS presentation-time viewporter)
                set(protocol_xml "${LIBREPAINT_WAYLAND_PROTOCOLS_DIR}/stable/${protocol_name}/${protocol_name}.xml")
                foreach(protocol_output IN ITEMS client-header private-code)
                    if(protocol_output STREQUAL "client-header")
                        set(protocol_file "${CMAKE_CURRENT_BINARY_DIR}/${protocol_name}-client-protocol.h")
                    else()
                        set(protocol_file "${CMAKE_CURRENT_BINARY_DIR}/${protocol_name}-protocol.c")
                    endif()
                    add_custom_command(OUTPUT "${protocol_file}"
                        COMMAND "${LIBREPAINT_WAYLAND_SCANNER}" ${protocol_output} "${protocol_xml}" "${protocol_file}"
                        DEPENDS "${protocol_xml}" VERBATIM)
                    target_sources(kritacanvasgpusurface PRIVATE "${protocol_file}")
                endforeach()
            endforeach()
            target_compile_definitions(kritacanvasgpusurface PRIVATE LIBREPAINT_GPU_WAYLAND)
            target_include_directories(kritacanvasgpusurface PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
            target_link_libraries(kritacanvasgpusurface PRIVATE Qt6::GuiPrivate PkgConfig::LIBREPAINT_WAYLAND_CLIENT)
        endif()
    endif()
endif()
